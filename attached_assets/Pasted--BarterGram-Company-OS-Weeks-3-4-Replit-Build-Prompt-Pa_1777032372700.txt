/**
 * googleSheets.ts
 * Writes KPI reports and P&L data to Google Sheets.
 * Uses the same Service Account as googleDrive.ts.
 */

import { google, sheets_v4 } from "googleapis";
import { JWT } from "google-auth-library";
import { getOrCreateSubFolder } from "./googleDrive.js";

function getAuth(): JWT {
  const credentials = JSON.parse(process.env.GOOGLE_SERVICE_ACCOUNT_JSON!);
  return new JWT({
    email: credentials.client_email,
    key: credentials.private_key,
    scopes: [
      "https://www.googleapis.com/auth/spreadsheets",
      "https://www.googleapis.com/auth/drive",
    ],
  });
}

function getSheets(): sheets_v4.Sheets {
  return google.sheets({ version: "v4", auth: getAuth() });
}

function getDrive() {
  return google.drive({ version: "v3", auth: getAuth() });
}

/**
 * Creates a new Google Sheet in the specified sub-folder and populates it.
 * Returns the sheet URL.
 */
export async function createReport(
  title: string,
  headers: string[],
  rows: (string | number | null)[][],
  subFolder: "finance" | "analytics" | "admin"
): Promise<string> {
  const sheets = getSheets();
  const drive = getDrive();

  // Create spreadsheet
  const spreadsheet = await sheets.spreadsheets.create({
    requestBody: {
      properties: { title },
      sheets: [{ properties: { title: "Report" } }],
    },
  });

  const sheetId = spreadsheet.data.spreadsheetId!;

  // Write headers + data
  const values = [headers, ...rows];
  await sheets.spreadsheets.values.update({
    spreadsheetId: sheetId,
    range: "Report!A1",
    valueInputOption: "RAW",
    requestBody: { values },
  });

  // Bold the header row
  await sheets.spreadsheets.batchUpdate({
    spreadsheetId: sheetId,
    requestBody: {
      requests: [
        {
          repeatCell: {
            range: { sheetId: 0, startRowIndex: 0, endRowIndex: 1 },
            cell: {
              userEnteredFormat: {
                textFormat: { bold: true },
                backgroundColor: { red: 0.2, green: 0.6, blue: 0.4 },
              },
            },
            fields: "userEnteredFormat(textFormat,backgroundColor)",
          },
        },
      ],
    },
  });

  // Move to correct Drive folder
  const folderNames = { finance: "Finance", analytics: "Analytics", admin: "Admin" };
  const folderId = await getOrCreateSubFolder(folderNames[subFolder]);
  const file = await drive.files.get({ fileId: sheetId, fields: "parents" });
  await drive.files.update({
    fileId: sheetId,
    addParents: folderId,
    removeParents: file.data.parents?.join(","),
    fields: "id",
  });

  return `https://docs.google.com/spreadsheets/d/${sheetId}/edit`;
}

/**
 * Appends rows to an existing sheet.
 */
export async function appendToSheet(
  spreadsheetId: string,
  rows: (string | number | null)[][]
): Promise<void> {
  const sheets = getSheets();
  await sheets.spreadsheets.values.append({
    spreadsheetId,
    range: "Report!A1",
    valueInputOption: "RAW",
    insertDataOption: "INSERT_ROWS",
    requestBody: { values: rows },
  });
}

/**
 * legalAgent.ts
 * Generates UAE-compliant barter contract templates,
 * reads dispute logs for risk summaries, and flags VAT/compliance issues.
 *
 * DISCLAIMER: All outputs are for guidance only — not legal advice.
 * Always recommend consulting a UAE-licensed legal professional.
 */

import OpenAI from "openai";
import { db } from "../../db.js";
import { legalDocuments } from "../../../shared/schema.js";
import { desc, eq, sql } from "drizzle-orm";
import { createGoogleDoc } from "../../connectors/googleDrive.js";
import { logAgentRun, isBudgetSafe } from "./costTracker.js";
import { notifyFounder } from "../../connectors/twilio.js";

const openai = new OpenAI();

const LEGAL_DISCLAIMER = `\n\n---\nDISCLAIMER: This document is generated for guidance purposes only and does not constitute legal advice. BarterGram is not a law firm. For legally binding agreements, consult a UAE-licensed legal professional. VAT guidance references UAE FTA regulations as of 2024 — verify current rates at uaevat.gov.ae.`;

// ─── Contract generation ──────────────────────────────────────────────────────

export interface ContractParams {
  partyAName: string;
  partyAOffer: string;
  partyAOfferValueAed: number;
  partyBName: string;
  partyBOffer: string;
  partyBOfferValueAed: number;
  city: string;
  dealId?: number;
}

/**
 * Generates a UAE barter contract and saves it to Google Drive.
 */
export async function generateBarterContract(params: ContractParams): Promise<{
  docUrl: string;
  riskLevel: string;
  vatNote: string;
}> {
  const { safe } = await isBudgetSafe();
  if (!safe) {
    return {
      docUrl: "",
      riskLevel: "unknown",
      vatNote: "AI budget exhausted — use manual template",
    };
  }

  const totalValueAed = params.partyAOfferValueAed + params.partyBOfferValueAed;
  const isHighValue = totalValueAed > 50000;
  const vatRequired = totalValueAed >= 375000; // UAE VAT registration threshold

  const systemPrompt = `You are a UAE legal document specialist for BarterGram, a barter marketplace.
Generate clear, professional barter agreements that reference UAE Commercial Transactions Law (Federal Decree-Law No. 50 of 2022).
Include: parties, offer descriptions, valuations in AED, delivery obligations, dispute resolution clause (BarterGram platform first, then Dubai Courts/DIFC), VAT note, and signature blocks.
Be specific and practical. Use formal but readable language.`;

  const userPrompt = `Generate a barter exchange agreement for:

Party A: ${params.partyAName}
Offers: ${params.partyAOffer} (estimated value: AED ${params.partyAOfferValueAed.toLocaleString()})

Party B: ${params.partyBName}
Offers: ${params.partyBOffer} (estimated value: AED ${params.partyBOfferValueAed.toLocaleString()})

Location: ${params.city}, UAE
Total deal value: AED ${totalValueAed.toLocaleString()}
High-value deal: ${isHighValue ? "Yes — include enhanced verification clause" : "No"}
VAT registration threshold applicable: ${vatRequired ? "Yes — include VAT advisory clause" : "No — values below AED 375,000 threshold"}

Include all standard UAE barter agreement clauses.`;

  const response = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: systemPrompt },
      { role: "user", content: userPrompt },
    ],
    max_tokens: 1500,
  });

  const contractContent = (response.choices[0].message.content ?? "") + LEGAL_DISCLAIMER;
  const title = `Barter Agreement — ${params.partyAName} & ${params.partyBName}`;
  const docUrl = await createGoogleDoc(title, contractContent, "legal");

  const riskLevel = isHighValue ? "high" : totalValueAed > 10000 ? "medium" : "low";
  const vatNote = vatRequired
    ? "⚠️ Deal value approaches VAT registration threshold. Advise parties to consult FTA."
    : "Deal value below UAE VAT registration threshold (AED 375,000).";

  await db.insert(legalDocuments).values({
    documentType: "barter_contract",
    title,
    content: contractContent.slice(0, 2000),
    googleDocUrl: docUrl,
    riskLevel,
    vatRelevant: vatRequired,
    status: "draft",
    notes: vatNote,
  });

  await logAgentRun({
    agentName: "legalAgent",
    triggeredBy: "api",
    input: `Contract: ${params.partyAName} <> ${params.partyBName}, AED ${totalValueAed}`,
    output: `Generated: ${docUrl}`,
    inputTokens: response.usage?.prompt_tokens ?? 0,
    outputTokens: response.usage?.completion_tokens ?? 0,
  });

  return { docUrl, riskLevel, vatNote };
}

// ─── Dispute risk summary ─────────────────────────────────────────────────────

/**
 * Reads recent dispute logs and generates a risk summary.
 * Runs weekly and whenever a high-risk dispute is escalated.
 */
export async function generateDisputeRiskSummary(): Promise<string> {
  const { safe } = await isBudgetSafe();
  if (!safe) return "⚠️ AI budget limit reached. Risk summary skipped.";

  // Read dispute data from existing ai_logs
  // Adjust to match your actual dispute table structure
  const disputeData = await db.execute(sql`
    SELECT
      COUNT(*) as total_disputes,
      SUM(CASE WHEN output ILIKE '%escalat%' THEN 1 ELSE 0 END) as escalated,
      SUM(CASE WHEN output ILIKE '%fraud%' OR output ILIKE '%scam%' THEN 1 ELSE 0 END) as fraud_flags,
      SUM(CASE WHEN output ILIKE '%vat%' OR output ILIKE '%tax%' THEN 1 ELSE 0 END) as vat_issues,
      SUM(CASE WHEN output ILIKE '%high.value%' OR output ILIKE '%50000%' THEN 1 ELSE 0 END) as high_value_disputes
    FROM company_os_logs
    WHERE agent_name = 'disputeResolution'
      AND created_at >= NOW() - INTERVAL '30 days'
  `);

  const stats = disputeData.rows[0] as {
    total_disputes: number;
    escalated: number;
    fraud_flags: number;
    vat_issues: number;
    high_value_disputes: number;
  };

  const systemPrompt = `You are a UAE legal risk analyst for BarterGram barter marketplace.
Write concise, actionable risk summaries. Reference UAE law where relevant.
Always add the disclaimer that this is guidance only, not legal advice.
Format for WhatsApp: use *bold* for key points, keep it under 200 words.`;

  const response = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: systemPrompt },
      {
        role: "user",
        content: `Generate a 30-day dispute risk summary for BarterGram UAE:
Total disputes: ${stats.total_disputes}
Escalated to admin: ${stats.escalated}
Fraud flags: ${stats.fraud_flags}
VAT-related issues: ${stats.vat_issues}
High-value disputes (>AED 50k): ${stats.high_value_disputes}

Identify the top 2-3 risk patterns and suggest one operational action for each.`,
      },
    ],
    max_tokens: 400,
  });

  const summary = (response.choices[0].message.content ?? "") + "\n\n_Guidance only — not legal advice._";

  // Save to Drive
  const title = `Legal Risk Summary — ${new Date().toLocaleDateString("en-AE", { timeZone: "Asia/Dubai" })}`;
  const docUrl = await createGoogleDoc(title, summary, "legal");

  await db.insert(legalDocuments).values({
    documentType: "dispute_summary",
    title,
    content: summary,
    googleDocUrl: docUrl,
    riskLevel: stats.fraud_flags > 2 ? "high" : stats.escalated > 5 ? "medium" : "low",
    status: "reviewed",
  });

  await logAgentRun({
    agentName: "legalAgent",
    triggeredBy: "scheduler",
    input: "monthly_risk_summary",
    output: summary,
    inputTokens: response.usage?.prompt_tokens ?? 0,
    outputTokens: response.usage?.completion_tokens ?? 0,
  });

  return `⚖️ *Legal Risk Summary*\n\n${summary}\n\n📄 Full report: ${docUrl}`;
}

// ─── VAT / compliance check ───────────────────────────────────────────────────

/**
 * Checks a deal value for UAE VAT implications and returns guidance.
 */
export async function checkVatCompliance(dealValueAed: number): Promise<string> {
  if (dealValueAed < 375000) {
    return `✅ Deal value (AED ${dealValueAed.toLocaleString()}) is below the UAE VAT registration threshold of AED 375,000. Standard barter rules apply.`;
  }

  if (dealValueAed >= 375000) {
    return [
      `⚠️ *VAT Advisory Notice*`,
      ``,
      `Deal value: AED ${dealValueAed.toLocaleString()}`,
      `UAE VAT registration threshold: AED 375,000`,
      ``,
      `This deal may trigger VAT obligations for one or both parties.`,
      `Both parties should verify their VAT registration status with the FTA (uaevat.gov.ae).`,
      `BarterGram platform fees are subject to 5% VAT.`,
      ``,
      `_For guidance only — consult a UAE tax professional._`,
    ].join("\n");
  }

  return "";
}

// ─── WhatsApp summary ─────────────────────────────────────────────────────────

export async function getLegalSummary(): Promise<string> {
  const recent = await db
    .select()
    .from(legalDocuments)
    .orderBy(desc(legalDocuments.createdAt))
    .limit(5);

  const highRisk = recent.filter((d) => d.riskLevel === "high").length;
  const vatDocs = recent.filter((d) => d.vatRelevant).length;

  const lines = [
    `⚖️ *Legal Agent Summary*`,
    ``,
    `Documents generated (recent): ${recent.length}`,
    `High-risk items: ${highRisk}`,
    `VAT-relevant documents: ${vatDocs}`,
    ``,
    `Recent documents:`,
    ...recent.slice(0, 3).map((d) => `• ${d.title} (${d.riskLevel} risk)`),
    ``,
    `_All outputs are for guidance only — not legal advice._`,
  ];

  return lines.join("\n");
}

/**
 * dashboardAgent.ts
 * Aggregates KPIs from all agents and tables.
 * Produces Recharts-ready JSON for the admin frontend.
 * Exports weekly reports to Google Sheets.
 */

import { db } from "../../db.js";
import {
  kpiSnapshots,
  financeSnapshots,
  campaignPerformance,
  salesLeads,
  companyOsLogs,
} from "../../../shared/schema.js";
import { desc, sql, gte } from "drizzle-orm";
import { createReport } from "../../connectors/googleSheets.js";
import { getMonthlySpendAed } from "./costTracker.js";
import { logAgentRun } from "./costTracker.js";
import { notifyFounder } from "../../connectors/twilio.js";

// ─── KPI aggregation ──────────────────────────────────────────────────────────

export interface DashboardKpis {
  // Users
  totalUsers: number;
  newUsersToday: number;
  activeUsers7d: number;
  // Posts & deals
  totalPosts: number;
  newPostsToday: number;
  totalDeals: number;
  completedDeals: number;
  dealCompletionRate: number;
  // Revenue
  gmvAed: number;
  revenueAed: number;
  revenueThisMonth: number;
  // Disputes
  openDisputes: number;
  resolvedDisputes: number;
  // Marketing
  activeCampaigns: number;
  totalAdSpendAed: number;
  // Costs
  aiCostThisMonthAed: number;
  aiCostBudgetAed: number;
  // Top metrics
  topCategory: string;
  topCity: string;
  // Chart data (Recharts format)
  revenueChart: Array<{ date: string; revenueAed: number; transactions: number }>;
  userGrowthChart: Array<{ date: string; totalUsers: number; newUsers: number }>;
  categoryChart: Array<{ category: string; count: number; gmvAed: number }>;
  agentActivityChart: Array<{ agent: string; runs: number; errors: number; costAed: number }>;
}

/**
 * Aggregates all KPIs from the database.
 * Adjust table/column names to match your actual schema.
 */
export async function aggregateKpis(): Promise<DashboardKpis> {
  const today = new Date().toISOString().split("T")[0];
  const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
  const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000);

  // Run all queries in parallel for speed
  const [
    userStats,
    postStats,
    dealStats,
    disputeStats,
    revenueHistory,
    campaignStats,
    categoryStats,
    agentStats,
    aiCost,
  ] = await Promise.all([
    // User stats — adjust table name as needed
    db.execute(sql`
      SELECT
        COUNT(*) as total,
        SUM(CASE WHEN created_at::date = CURRENT_DATE THEN 1 ELSE 0 END) as today,
        SUM(CASE WHEN created_at >= ${sevenDaysAgo.toISOString()} THEN 1 ELSE 0 END) as active_7d
      FROM users
    `),

    // Post stats
    db.execute(sql`
      SELECT
        COUNT(*) as total,
        SUM(CASE WHEN created_at::date = CURRENT_DATE THEN 1 ELSE 0 END) as today
      FROM barter_posts
      WHERE status != 'deleted'
    `),

    // Deal stats + GMV
    db.execute(sql`
      SELECT
        COUNT(*) as total,
        SUM(CASE WHEN status = 'completed' THEN 1 ELSE 0 END) as completed,
        COALESCE(SUM(CASE WHEN status = 'completed' THEN estimated_value_aed ELSE 0 END), 0) as gmv_aed
      FROM barter_deals
    `),

    // Dispute stats
    db.execute(sql`
      SELECT
        SUM(CASE WHEN status = 'open' THEN 1 ELSE 0 END) as open_count,
        SUM(CASE WHEN status = 'resolved' THEN 1 ELSE 0 END) as resolved_count
      FROM barter_disputes
    `),

    // Revenue from finance snapshots (last 30 days)
    db.select()
      .from(financeSnapshots)
      .where(gte(financeSnapshots.snapshotDate, thirtyDaysAgo.toISOString().split("T")[0]))
      .orderBy(desc(financeSnapshots.snapshotDate))
      .limit(30),

    // Campaign stats
    db.select()
      .from(campaignPerformance)
      .orderBy(desc(campaignPerformance.fetchedAt))
      .limit(20),

    // Category breakdown
    db.execute(sql`
      SELECT
        category,
        COUNT(*) as count,
        COALESCE(SUM(estimated_value_aed), 0) as gmv_aed,
        city
      FROM barter_posts
      WHERE created_at >= ${thirtyDaysAgo.toISOString()}
        AND status = 'active'
      GROUP BY category, city
      ORDER BY count DESC
      LIMIT 10
    `),

    // Agent activity
    db.execute(sql`
      SELECT
        agent_name,
        COUNT(*) as runs,
        SUM(CASE WHEN success = false THEN 1 ELSE 0 END) as errors,
        COALESCE(SUM(cost_aed::numeric), 0) as cost_aed
      FROM company_os_logs
      WHERE created_at >= ${thirtyDaysAgo.toISOString()}
      GROUP BY agent_name
      ORDER BY runs DESC
    `),

    getMonthlySpendAed(),
  ]);

  const users = userStats.rows[0] as { total: number; today: number; active_7d: number };
  const posts = postStats.rows[0] as { total: number; today: number };
  const deals = dealStats.rows[0] as { total: number; completed: number; gmv_aed: number };
  const disputes = disputeStats.rows[0] as { open_count: number; resolved_count: number };
  const categories = categoryStats.rows as Array<{ category: string; count: number; gmv_aed: number; city: string }>;
  const agentActivity = agentStats.rows as Array<{ agent_name: string; runs: number; errors: number; cost_aed: number }>;

  const revenueThisMonth = revenueHistory.reduce(
    (sum, r) => sum + parseFloat(String(r.revenueAed ?? "0")), 0
  );

  const activeCampaigns = campaignStats.filter((c) => c.status === "ACTIVE").length;
  const totalAdSpend = campaignStats.reduce(
    (sum, c) => sum + parseFloat(String(c.spendAed ?? "0")), 0
  );

  const topCategory = categories[0]?.category ?? "General";
  const topCity = categories[0]?.city ?? "Dubai";

  const completionRate = deals.total > 0
    ? parseFloat(((deals.completed / deals.total) * 100).toFixed(1))
    : 0;

  // Build chart data
  const revenueChart = revenueHistory.slice(0, 14).reverse().map((r) => ({
    date: r.snapshotDate,
    revenueAed: parseFloat(String(r.revenueAed ?? "0")),
    transactions: r.transactionCount ?? 0,
  }));

  const categoryChart = categories.slice(0, 8).map((c) => ({
    category: c.category,
    count: Number(c.count),
    gmvAed: parseFloat(String(c.gmv_aed ?? "0")),
  }));

  const agentActivityChart = agentActivity.map((a) => ({
    agent: a.agent_name.replace("Agent", ""),
    runs: Number(a.runs),
    errors: Number(a.errors),
    costAed: parseFloat(String(a.cost_aed ?? "0")),
  }));

  return {
    totalUsers: Number(users.total ?? 0),
    newUsersToday: Number(users.today ?? 0),
    activeUsers7d: Number(users.active_7d ?? 0),
    totalPosts: Number(posts.total ?? 0),
    newPostsToday: Number(posts.today ?? 0),
    totalDeals: Number(deals.total ?? 0),
    completedDeals: Number(deals.completed ?? 0),
    dealCompletionRate: completionRate,
    gmvAed: parseFloat(String(deals.gmv_aed ?? "0")),
    revenueAed: revenueHistory[0] ? parseFloat(String(revenueHistory[0].revenueAed ?? "0")) : 0,
    revenueThisMonth,
    openDisputes: Number(disputes.open_count ?? 0),
    resolvedDisputes: Number(disputes.resolved_count ?? 0),
    activeCampaigns,
    totalAdSpendAed: totalAdSpend,
    aiCostThisMonthAed: aiCost,
    aiCostBudgetAed: parseFloat(process.env.COMPANY_OS_MONTHLY_BUDGET_AED ?? "400"),
    topCategory,
    topCity,
    revenueChart,
    userGrowthChart: [], // populate from user history table if available
    categoryChart,
    agentActivityChart,
  };
}

// ─── Daily KPI snapshot ───────────────────────────────────────────────────────

export async function saveDailyKpiSnapshot(): Promise<DashboardKpis> {
  const kpis = await aggregateKpis();
  const today = new Date().toISOString().split("T")[0];

  await db
    .insert(kpiSnapshots)
    .values({
      snapshotDate: today,
      snapshotType: "daily",
      totalUsers: kpis.totalUsers,
      newUsersToday: kpis.newUsersToday,
      activeUsers7d: kpis.activeUsers7d,
      totalPosts: kpis.totalPosts,
      newPostsToday: kpis.newPostsToday,
      totalDeals: kpis.totalDeals,
      completedDeals: kpis.completedDeals,
      dealCompletionRate: kpis.dealCompletionRate.toFixed(2),
      gmvAed: kpis.gmvAed.toFixed(2),
      revenueAed: kpis.revenueAed.toFixed(2),
      openDisputes: kpis.openDisputes,
      resolvedDisputes: kpis.resolvedDisputes,
      topCategory: kpis.topCategory,
      topCity: kpis.topCity,
      aiCostAed: kpis.aiCostThisMonthAed.toFixed(2),
    })
    .onConflictDoNothing();

  await logAgentRun({
    agentName: "dashboardAgent",
    triggeredBy: "scheduler",
    input: "daily_kpi_snapshot",
    output: `KPIs saved: ${kpis.totalUsers} users, AED ${kpis.revenueAed} revenue`,
    inputTokens: 0,
    outputTokens: 0,
  });

  return kpis;
}

export async function exportWeeklyReport(): Promise<string> {
  const kpis = await aggregateKpis();

  const headers = [
    "Date", "Total Users", "New Users", "Active 7d",
    "Total Posts", "Deals", "Completed Deals", "Completion Rate %",
    "GMV AED", "Revenue AED", "Open Disputes", "Top Category", "AI Cost AED"
  ];

  const today = new Date().toLocaleDateString("en-AE", { timeZone: "Asia/Dubai" });

  const rows = [[
    today,
    kpis.totalUsers,
    kpis.newUsersToday,
    kpis.activeUsers7d,
    kpis.totalPosts,
    kpis.totalDeals,
    kpis.completedDeals,
    kpis.dealCompletionRate,
    kpis.gmvAed,
    kpis.revenueThisMonth,
    kpis.openDisputes,
    kpis.topCategory,
    kpis.aiCostThisMonthAed,
  ]];

  // Add revenue history rows
  for (const r of kpis.revenueChart) {
    rows.push([r.date, null, null, null, null, null, null, null, null, r.revenueAed, null, null, null]);
  }

  const title = `BarterGram Weekly Report — W${getWeekNumber()}`;
  const sheetUrl = await createReport(title, headers, rows, "analytics");

  await notifyFounder(
    `📊 *Weekly Report Ready*\n\nKPIs exported to Google Sheets:\n${sheetUrl}\n\nHighlights:\nUsers: ${kpis.totalUsers} (+${kpis.newUsersToday} today)\nRevenue this month: AED ${kpis.revenueThisMonth.toFixed(0)}\nDeal completion: ${kpis.dealCompletionRate}%\nGMV: AED ${kpis.gmvAed.toFixed(0)}`
  );

  return sheetUrl;
}

function getWeekNumber(): string {
  const now = new Date();
  const week = Math.ceil(((now.getTime() - new Date(now.getFullYear(), 0, 1).getTime()) / 86400000 + 1) / 7);
  return `${now.getFullYear()}-${String(week).padStart(2, "0")}`;
}
