```typescript
// Monthly spend limits per agent in AED
const AGENT_LIMITS_AED: Record<string, number> = {
  managerAgent: 60,
  marketingAgent: 80,
  salesAgent: 60,
  legalAgent: 50,
  dashboardAgent: 20,
  financeAgent: 10,
  intelligenceAgent: 80,
  boardReportAgent: 40,
  // Product agents share the rest
  default: 30,
};

export async function getAgentMonthlySpend(agentName: string): Promise<number> {
  const startOfMonth = new Date();
  startOfMonth.setDate(1);
  startOfMonth.setHours(0, 0, 0, 0);

  const result = await db
    .select({ total: sql<string>`COALESCE(SUM(cost_aed::numeric), 0)` })
    .from(companyOsLogs)
    .where(
      sql`agent_name = ${agentName} AND created_at >= ${startOfMonth.toISOString()}`
    );

  return parseFloat(result[0]?.total ?? "0");
}

export async function isAgentBudgetSafe(agentName: string): Promise<boolean> {
  const spent = await getAgentMonthlySpend(agentName);
  const limit = AGENT_LIMITS_AED[agentName] ?? AGENT_LIMITS_AED.default;
  return spent < limit * 0.95;
}

export async function getAllAgentSpends(): Promise<Array<{
  agentName: string;
  spentAed: number;
  limitAed: number;
  pct: number;
}>> {
  const startOfMonth = new Date();
  startOfMonth.setDate(1);
  startOfMonth.setHours(0, 0, 0, 0);

  const rows = await db
    .select({
      agentName: companyOsLogs.agentName,
      total: sql<string>`COALESCE(SUM(cost_aed::numeric), 0)`,
    })
    .from(companyOsLogs)
    .where(gte(companyOsLogs.createdAt, startOfMonth))
    .groupBy(companyOsLogs.agentName);

  return rows.map((r) => {
    const spent = parseFloat(r.total ?? "0");
    const limit = AGENT_LIMITS_AED[r.agentName] ?? AGENT_LIMITS_AED.default;
    return {
      agentName: r.agentName,
      spentAed: spent,
      limitAed: limit,
      pct: parseFloat(((spent / limit) * 100).toFixed(1)),
    };
  });
}
```

```typescript
/**
 * memoryAgent.ts
 * Shared memory layer for all Company OS agents.
 * Agents write learnings here. Other agents read them before acting.
 * This is how agents "learn from each other" without direct coupling.
 */

import { db } from "../../db.js";
import { agentMemory } from "../../../shared/schema.js";
import { eq, desc, sql, and, gt } from "drizzle-orm";

export type MemoryType = "insight" | "pattern" | "preference" | "alert";

// ─── Write memory ─────────────────────────────────────────────────────────────

/**
 * Stores or updates a memory entry.
 * Use this after any agent discovers something worth remembering.
 */
export async function remember(params: {
  key: string;
  type: MemoryType;
  sourceAgent: string;
  value: string | object;
  confidence?: number;
  expiresInDays?: number;
}): Promise<void> {
  const valueStr = typeof params.value === "object"
    ? JSON.stringify(params.value)
    : params.value;

  const expiresAt = params.expiresInDays
    ? new Date(Date.now() + params.expiresInDays * 86400000)
    : null;

  // Upsert by memory key
  const existing = await db
    .select()
    .from(agentMemory)
    .where(eq(agentMemory.memoryKey, params.key))
    .limit(1);

  if (existing.length > 0) {
    await db
      .update(agentMemory)
      .set({
        value: valueStr,
        sourceAgent: params.sourceAgent,
        confidence: (params.confidence ?? 0.8).toString(),
        updatedAt: new Date(),
        expiresAt,
      })
      .where(eq(agentMemory.memoryKey, params.key));
  } else {
    await db.insert(agentMemory).values({
      memoryKey: params.key,
      memoryType: params.type,
      sourceAgent: params.sourceAgent,
      value: valueStr,
      confidence: (params.confidence ?? 0.8).toString(),
      expiresAt,
    });
  }
}

// ─── Read memory ──────────────────────────────────────────────────────────────

/**
 * Retrieves a memory value by key.
 * Returns null if not found or expired.
 */
export async function recall(key: string): Promise<string | null> {
  const rows = await db
    .select()
    .from(agentMemory)
    .where(
      and(
        eq(agentMemory.memoryKey, key),
        sql`(expires_at IS NULL OR expires_at > NOW())`
      )
    )
    .limit(1);

  if (!rows.length) return null;

  // Increment access counter
  await db
    .update(agentMemory)
    .set({ timesAccessed: (rows[0].timesAccessed ?? 0) + 1 })
    .where(eq(agentMemory.memoryKey, key));

  return rows[0].value;
}

/**
 * Retrieves all memories of a given type.
 * Useful for agents to load context before acting.
 */
export async function recallByType(
  type: MemoryType,
  limit = 20
): Promise<Array<{ key: string; value: string; sourceAgent: string; confidence: number }>> {
  const rows = await db
    .select()
    .from(agentMemory)
    .where(
      and(
        eq(agentMemory.memoryType, type),
        sql`(expires_at IS NULL OR expires_at > NOW())`
      )
    )
    .orderBy(desc(agentMemory.updatedAt))
    .limit(limit);

  return rows.map((r) => ({
    key: r.memoryKey,
    value: r.value,
    sourceAgent: r.sourceAgent,
    confidence: parseFloat(String(r.confidence ?? "0.8")),
  }));
}

/**
 * Builds a context string from all current insights.
 * Agents include this in their system prompts for better output.
 */
export async function buildAgentContext(agentName: string): Promise<string> {
  const insights = await recallByType("insight", 10);
  const patterns = await recallByType("pattern", 5);

  if (!insights.length && !patterns.length) return "";

  const lines = [
    "CURRENT PLATFORM INTELLIGENCE (from shared agent memory):",
    ...insights.slice(0, 5).map((i) => `- ${i.key}: ${i.value}`),
    ...patterns.slice(0, 3).map((p) => `- Pattern: ${p.value}`),
  ];

  return lines.join("\n");
}
```

```typescript
/**
 * intelligenceAgent.ts
 * Proactive intelligence — detects opportunities and risks without being asked.
 * Runs on a schedule and pushes alerts to WhatsApp.
 * Reads from shared memory + all agent logs.
 */

import OpenAI from "openai";
import { db } from "../../db.js";
import { proactiveAlerts, kpiSnapshots, financeSnapshots } from "../../../shared/schema.js";
import { desc, eq, sql, and } from "drizzle-orm";
import { recallByType, buildAgentContext } from "./memoryAgent.js";
import { logAgentRun, isAgentBudgetSafe } from "./costTracker.js";
import { notifyFounder } from "../../connectors/twilio.js";

const openai = new OpenAI();

// ─── Anomaly detection ────────────────────────────────────────────────────────

/**
 * Compares today's KPIs against the 7-day average.
 * Flags significant deviations as alerts.
 */
async function detectAnomalies(): Promise<Array<{ type: string; message: string; priority: "low" | "medium" | "high" | "critical" }>> {
  const alerts: Array<{ type: string; message: string; priority: "low" | "medium" | "high" | "critical" }> = [];

  const recentSnapshots = await db
    .select()
    .from(financeSnapshots)
    .orderBy(desc(financeSnapshots.snapshotDate))
    .limit(8);

  if (recentSnapshots.length < 3) return alerts;

  const today = recentSnapshots[0];
  const previous7 = recentSnapshots.slice(1);

  const avgRevenue = previous7.reduce((s, r) => s + parseFloat(String(r.revenueAed ?? "0")), 0) / previous7.length;
  const avgTransactions = previous7.reduce((s, r) => s + (r.transactionCount ?? 0), 0) / previous7.length;

  const todayRevenue = parseFloat(String(today.revenueAed ?? "0"));
  const todayTransactions = today.transactionCount ?? 0;

  // Revenue spike — positive anomaly
  if (todayRevenue > avgRevenue * 1.5 && todayRevenue > 100) {
    alerts.push({
      type: "opportunity",
      message: `🚀 Revenue spike detected! Today: AED ${todayRevenue.toFixed(0)} vs 7-day avg: AED ${avgRevenue.toFixed(0)} (+${(((todayRevenue / avgRevenue) - 1) * 100).toFixed(0)}%). Consider increasing ad spend in Zeely.`,
      priority: "high",
    });
  }

  // Revenue drop — risk
  if (todayRevenue < avgRevenue * 0.4 && avgRevenue > 50) {
    alerts.push({
      type: "risk",
      message: `⚠️ Revenue drop: AED ${todayRevenue.toFixed(0)} today vs AED ${avgRevenue.toFixed(0)} average. Check Stripe for payment issues.`,
      priority: "high",
    });
  }

  // Transaction volume anomaly
  if (todayTransactions > avgTransactions * 2 && todayTransactions > 5) {
    alerts.push({
      type: "opportunity",
      message: `📈 Transaction volume 2x normal. ${todayTransactions} transactions today vs avg ${avgTransactions.toFixed(0)}. Platform engagement is high.`,
      priority: "medium",
    });
  }

  return alerts;
}

/**
 * Checks KPI snapshots for platform health patterns.
 */
async function detectPlatformPatterns(): Promise<Array<{ type: string; message: string; priority: "low" | "medium" | "high" | "critical" }>> {
  const alerts: Array<{ type: string; message: string; priority: "low" | "medium" | "high" | "critical" }> = [];

  const latest = await db
    .select()
    .from(kpiSnapshots)
    .orderBy(desc(kpiSnapshots.snapshotDate))
    .limit(7);

  if (!latest.length) return alerts;

  const now = latest[0];
  const openDisputes = now.openDisputes ?? 0;
  const completionRate = parseFloat(String(now.dealCompletionRate ?? "0"));
  const aiCostAed = parseFloat(String(now.aiCostAed ?? "0"));
  const aiCostBudget = parseFloat(process.env.COMPANY_OS_MONTHLY_BUDGET_AED ?? "400");

  if (openDisputes >= 10) {
    alerts.push({
      type: "risk",
      message: `🔴 ${openDisputes} open disputes — above normal threshold. Review dispute queue in /admin. Consider running dispute risk report (reply 'risk report').`,
      priority: "critical",
    });
  } else if (openDisputes >= 5) {
    alerts.push({
      type: "risk",
      message: `⚠️ ${openDisputes} open disputes. Elevated but manageable. Legal Agent risk report recommended.`,
      priority: "medium",
    });
  }

  if (completionRate < 40 && Number(now.totalDeals ?? 0) > 20) {
    alerts.push({
      type: "risk",
      message: `⚠️ Deal completion rate at ${completionRate}% — below 40% benchmark. Users may be abandoning deals. Engagement Agent should escalate nudges.`,
      priority: "high",
    });
  }

  if (aiCostAed > aiCostBudget * 0.8) {
    alerts.push({
      type: "risk",
      message: `🤖 AI cost at AED ${aiCostAed.toFixed(0)} — 80% of AED ${aiCostBudget} monthly budget. Review agent usage with 'costs' command.`,
      priority: "high",
    });
  }

  // Milestone detection
  const totalUsers = now.totalUsers ?? 0;
  const milestones = [100, 250, 500, 1000, 2500, 5000, 10000];
  for (const milestone of milestones) {
    const prev = latest[1]?.totalUsers ?? 0;
    if (totalUsers >= milestone && prev < milestone) {
      alerts.push({
        type: "milestone",
        message: `🎉 Milestone reached: ${milestone} users on BarterGram! Consider a celebration post via 'draft post milestone ${milestone} users'.`,
        priority: "high",
      });
    }
  }

  return alerts;
}

// ─── AI-powered opportunity detection ────────────────────────────────────────

/**
 * Uses GPT to analyse patterns in shared memory and identify growth opportunities.
 * Runs weekly — uses AI so checks budget first.
 */
export async function detectOpportunities(): Promise<string> {
  const canRun = await isAgentBudgetSafe("intelligenceAgent");
  if (!canRun) return "Budget limit — opportunity scan skipped.";

  const insights = await recallByType("insight", 15);
  const patterns = await recallByType("pattern", 10);
  const context = await buildAgentContext("intelligenceAgent");

  if (!insights.length) return "Insufficient data for opportunity detection yet.";

  const response = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      {
        role: "system",
        content: `You are the BarterGram growth intelligence analyst. 
Analyse platform data and identify 2-3 specific, actionable growth opportunities for a UAE barter marketplace.
Be specific: name categories, cities, user types, and suggest exact actions.
Format for WhatsApp: use *bold*, keep total under 200 words.
Focus on: underserved categories, geographic expansion, user activation, revenue optimisation.`,
      },
      {
        role: "user",
        content: `Analyse this BarterGram data and identify growth opportunities:\n\n${context}\n\nRecent patterns:\n${patterns.map((p) => `- ${p.key}: ${p.value}`).join("\n")}\n\nProvide 2-3 specific opportunities the founder can act on this week.`,
      },
    ],
    max_tokens: 350,
  });

  const output = response.choices[0].message.content ?? "";

  await logAgentRun({
    agentName: "intelligenceAgent",
    triggeredBy: "scheduler",
    input: "opportunity_scan",
    output,
    inputTokens: response.usage?.prompt_tokens ?? 0,
    outputTokens: response.usage?.completion_tokens ?? 0,
  });

  return output;
}

// ─── Main intelligence run ────────────────────────────────────────────────────

/**
 * Full intelligence run — anomalies + patterns + AI opportunities.
 * Called by scheduler twice daily.
 */
export async function runIntelligenceScan(): Promise<void> {
  const [anomalies, platformAlerts] = await Promise.all([
    detectAnomalies(),
    detectPlatformPatterns(),
  ]);

  const allAlerts = [...anomalies, ...platformAlerts];

  for (const alert of allAlerts) {
    // Store alert
    await db.insert(proactiveAlerts).values({
      alertType: alert.type,
      priority: alert.priority,
      title: alert.type.charAt(0).toUpperCase() + alert.type.slice(1),
      message: alert.message,
      sourceAgent: "intelligenceAgent",
      dataSnapshot: JSON.stringify(alert),
      sentToFounder: alert.priority === "high" || alert.priority === "critical",
    });

    // Send high/critical alerts immediately to WhatsApp
    if (alert.priority === "high" || alert.priority === "critical") {
      await notifyFounder(alert.message);
    }
  }

  // Log the scan
  await logAgentRun({
    agentName: "intelligenceAgent",
    triggeredBy: "scheduler",
    input: "intelligence_scan",
    output: `Found ${allAlerts.length} alerts (${allAlerts.filter((a) => a.priority === "high" || a.priority === "critical").length} high priority)`,
    inputTokens: 0,
    outputTokens: 0,
  });
}

export async function getAlertsSummary(): Promise<string> {
  const recent = await db
    .select()
    .from(proactiveAlerts)
    .where(sql`created_at >= NOW() - INTERVAL '7 days'`)
    .orderBy(desc(proactiveAlerts.createdAt))
    .limit(10);

  if (!recent.length) {
    return "✅ No alerts in the last 7 days. Platform looks healthy.";
  }

  const critical = recent.filter((a) => a.priority === "critical").length;
  const high = recent.filter((a) => a.priority === "high").length;
  const opportunities = recent.filter((a) => a.alertType === "opportunity").length;

  const topAlerts = recent.slice(0, 3).map((a) => `• ${a.message.slice(0, 100)}...`).join("\n");

  return [
    `🧠 *Intelligence Summary (7 days)*`,
    ``,
    `Critical: ${critical} | High: ${high} | Opportunities: ${opportunities}`,
    ``,
    `Recent alerts:`,
    topAlerts,
  ].join("\n");
}
```

```typescript
/**

 * Generates a monthly "board report" — executive summary of company performance.
 * Saved to Google Drive as a formatted Doc. Sent to founder via WhatsApp.
 */

import OpenAI from "openai";
import { db } from "../../db.js";
import {
  boardReports,
  kpiSnapshots,
  financeSnapshots,
  campaignPerformance,
  salesLeads,
} from "../../../shared/schema.js";
import { desc, sql, gte } from "drizzle-orm";
import { createGoogleDoc } from "../../connectors/googleDrive.js";
import { buildAgentContext } from "./memoryAgent.js";
import { logAgentRun, isAgentBudgetSafe } from "./costTracker.js";
import { notifyFounder } from "../../connectors/twilio.js";

const openai = new OpenAI();

export async function generateMonthlyBoardReport(): Promise<string> {
  const canRun = await isAgentBudgetSafe("boardReportAgent");
  if (!canRun) return "⚠️ AI budget limit — board report skipped. Will retry next check.";

  const now = new Date();
  const reportMonth = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}`;
  const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);

  // Gather all data for the month
  const [financeData, kpiData, campaignData, leadData, memoryContext] = await Promise.all([
    db.select().from(financeSnapshots)
      .where(gte(financeSnapshots.snapshotDate, monthStart.toISOString().split("T")[0]))
      .orderBy(desc(financeSnapshots.snapshotDate)),

    db.select().from(kpiSnapshots)
      .where(gte(kpiSnapshots.snapshotDate, monthStart.toISOString().split("T")[0]))
      .orderBy(desc(kpiSnapshots.snapshotDate))
      .limit(1),

    db.select().from(campaignPerformance)
      .where(gte(campaignPerformance.fetchedAt, monthStart))
      .orderBy(desc(campaignPerformance.spendAed))
      .limit(5),

    db.execute(sql`
      SELECT
        COUNT(*) as total_new,
        SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) as converted,
        SUM(CASE WHEN status = 're-engaged' THEN 1 ELSE 0 END) as re_engaged
      FROM sales_leads
      WHERE created_at >= ${monthStart.toISOString()}
    `),

    buildAgentContext("boardReportAgent"),
  ]);

  const totalRevenueAed = financeData.reduce(
    (s, r) => s + parseFloat(String(r.revenueAed ?? "0")), 0
  );

  const latestKpi = kpiData[0];
  const leads = leadData.rows[0] as { total_new: number; converted: number; re_engaged: number };

  const totalAdSpend = campaignData.reduce(
    (s, c) => s + parseFloat(String(c.spendAed ?? "0")), 0
  );

  const monthName = now.toLocaleString("en-AE", { month: "long", timeZone: "Asia/Dubai" });

  const dataContext = `
MONTH: ${monthName} ${now.getFullYear()}

FINANCIAL:
```

```typescript
// ─── Self-healing retry wrapper ───────────────────────────────────────────────

/**
 * Wraps any agent function with retry logic and error reporting.
 * If it fails twice, notifies the founder via WhatsApp.
 */
export async function withRetry<T>(
  fn: () => Promise<T>,
  agentName: string,
  taskName: string,
  maxRetries = 2
): Promise<T | null> {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      const errMsg = error instanceof Error ? error.message : String(error);
      console.error(`[${agentName}] ${taskName} failed (attempt ${attempt}/${maxRetries}):`, errMsg);

      await logAgentRun({
        agentName,
        triggeredBy: "retry",
        input: taskName,
        output: `ERROR attempt ${attempt}: ${errMsg}`,
        inputTokens: 0,
        outputTokens: 0,
        success: false,
      });

      if (attempt === maxRetries) {
        // Final failure — alert founder
        try {
          const { notifyFounder } = await import("../connectors/twilio.js");
          await notifyFounder(
            `🔴 *Agent Error*\nAgent: ${agentName}\nTask: ${taskName}\nError: ${errMsg.slice(0, 150)}\n\nCheck /admin/company-os for details.`
          );
        } catch {
          console.error("Failed to send error notification to founder");
        }
        return null;
      }

      // Wait before retry (exponential backoff)
      await new Promise((resolve) => setTimeout(resolve, 1000 * attempt));
    }
  }
  return null;
}
```
